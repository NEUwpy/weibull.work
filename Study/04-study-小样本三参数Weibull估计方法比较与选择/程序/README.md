```powershell
python/.venv/Scripts/python.exe "Study/04-study-小样本三参数Weibull估计方法比较与选择/程序/运行参数估计.py" --beta 5 --n 7 --output "Study/04-study-小样本三参数Weibull估计方法比较与选择/结果/复算/β5-n7"
```
