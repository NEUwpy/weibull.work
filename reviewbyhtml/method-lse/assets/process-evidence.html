  32 def _log_weibull_order_stat_mean(i, n):
  33     """E[W_(i:n)]，W 服从 reduced Log-Weibull 密度 h(w)=exp(w-e^w)（论文式(5)）。
  34
  35     对数密度积分（w 域），避免二项式交替求和在 n>=30 时的灾难性抵消。
  36     """
  37     log_c = special.gammaln(n + 1) - special.gammaln(i) - special.gammaln(n - i + 1)
  38
  39     def integrand(w):
  40         ew = np.exp(w)
  41         log_h = w - ew
  42         log_cdf = np.log(-np.expm1(-ew)) if ew < 700 else 0.0
  43         log_pdf = log_c + (i - 1) * log_cdf + (n - i) * (-ew) + log_h
  44         return w * np.exp(log_pdf)
  45
  46     guess = math.log(-math.log(1.0 - i / (n + 1.0)))
  47     value, _ = integrate.quad(integrand, -60.0, 10.0, limit=400, points=[guess])
  48     return value

 115         def white_fit(mu):
 116             """返回 (slope, intercept, F)；无效时返回 None。"""
 117             shifted = t - mu
 118             if np.any(shifted <= 0):
 119                 return None
 120             y = np.log(shifted)
 121             y_bar = y.mean()
 122             # 退化保护：y 的相对极差低于浮点噪声级别时视为无信息样本
 123             if float(y[-1] - y[0]) <= max(abs(float(y_bar)), 1.0) * 1e-12:
 124                 return None
 125             sy2_total = float(np.sum((y - y_bar) ** 2))
 126             if sxx <= 0 or sy2_total <= 0:
 127                 return None
 128             slope = float(np.sum((X - x_bar) * (y - y_bar))) / sxx
 129             if slope <= 0:
 130                 return None
 131             intercept = y_bar - slope * x_bar
 132             residuals = y - (intercept + slope * X)
 133             s_y2 = sy2_total / (n - 1)
 134             s_res2 = float(np.sum(residuals ** 2)) / (n - 2)
 135             f_ratio = float('inf') if s_res2 <= 0 else s_y2 / s_res2
 136             if trace:
 137                 evaluations.append(dict(x=float(mu), y=f_ratio, evaluation=len(evaluations) + 1))
 138             return slope, intercept, f_ratio

  79     def _calculate_r2(self, beta, eta, gamma_val):
  80         """Helper to calculate R-squared for any method"""
  81         if gamma_val >= self.data[0]: return 0.0
  82
  83         # Experimental
  84         x = np.log(self.data - gamma_val)
  85         y_exp = np.log(-np.log(1 - self._median_ranks()))
  86
  87         # Theoretical / Predicted
  88         # y = beta * x - beta * ln(eta)
  89         y_pred = beta * x - beta * np.log(eta)
  90
  91         # R^2 calculation
  92         ss_res = np.sum((y_exp - y_pred)**2)
  93         ss_tot = np.sum((y_exp - np.mean(y_exp))**2)
  94
  95         if ss_tot == 0: return 0.0
  96         return 1 - (ss_res / ss_tot)

  58     def _median_ranks(self):
  59         """
  60         Helper: Calculate Median Ranks using configured method.
  61         - 'bernard': Bernard's approximation (i - 0.3) / (n + 0.4)
  62         - 'exact': Exact median rank using inverse incomplete beta function
  63         """
  64         i = np.arange(1, self.n + 1)
  65         if self.rank_method == 'exact':
  66             return betaincinv(i, self.n - i + 1, 0.5)
  67         else:  # bernard
  68             return (i - 0.3) / (self.n + 0.4)
