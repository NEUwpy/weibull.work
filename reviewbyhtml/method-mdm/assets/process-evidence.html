  79     def _calculate_r2(self, beta, eta, gamma_val):
  80         """Helper to calculate R-squared for any method"""
  81         if gamma_val >= self.data[0]: return 0.0
  82
  83         # Experimental
  84         x = np.log(self.data - gamma_val)
  85         y_exp = np.log(-np.log(1 - self._median_ranks()))
  86
  87         # Theoretical / Predicted
  88         # y = beta * x - beta * ln(eta)
  89         y_pred = beta * x - beta * np.log(eta)
  90
  91         # R^2 calculation
  92         ss_res = np.sum((y_exp - y_pred)**2)

  96         return 1 - (ss_res / ss_tot)

 499                     "best_beta": float(b),
 500                     "best_eta": eta_mean,
 501                     "source": source,
 502                 }
 503                 if virtual:
 504                     point["virtual"] = True
 505                 grad_gamma_points[gamma] = point
 506
 507             for i in range(len(gammas)):
 508                 add_gradient_point(gammas[i], grads[i])
 509
 510             if solution_strategy == "brent_root":
 511                 if root_solver == "right_edge_fit":
 512                     add_gradient_point(found_gamma, offset, source="solver_root", virtual=True)
 513                 else:
 514                     add_gradient_point(found_gamma, profile_gradient(found_gamma), source="solver_root")
 515             elif solution_strategy == "truncated_at_zero":
 516                 add_gradient_point(0.0, probe_gradient_at_zero, source="solver_root")
 517
 518             grad_gamma_curve = sorted(
 519                 grad_gamma_points.values(),
 520                 key=lambda point: point["gamma"],
 521                 reverse=True,
 522             )
 523
 524             self.trace_data = {
 525                 "sigma_beta_curve": sigma_beta_curve,
 526                 "grad_gamma_curve": grad_gamma_curve,
 527                 "sigma_beta_gamma": sigma_beta_gamma,
 528                 **self.last_solution_info,
 529             }
