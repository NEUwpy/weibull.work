参数估计方法 / MLE

极大似然估计 MLE

不断尝试位置γ；每到一个位置，都求对应的形状β和尺度η。再比较合格候选的似然，确定最终三参数。

每换一个候选位置 γ，就重复条件形状与尺度求解；最后比较合格位置的似然。

数据预处理

进入求解器，以最小寿命u归一化样本。返回候选时恢复原单位。

  61  def mle_profile(data, gamma, *, shape_lower=1.0, shape_evaluations=None):
  62      """Maximize the standard likelihood at fixed gamma over beta>=shape_lower.
  63
  64      shape_lower=1 is the R00 diagnostic domain. shape_lower=0 is an explicitly
  65      unrestricted diagnostic (also needed by the distinct MMLE construction).
  66      At beta=1 the scale is recalculated; the shape score need not be zero.
  67      """
  68      if shape_lower < 0 or not np.isfinite(shape_lower):
  69          raise ValueError("shape_lower must be finite and nonnegative")
  70      d = _distances(data, gamma)
  71      t = np.log(d)
  72      solution = _shape(t, lower=shape_lower, evaluations=shape_evaluations)
  73      if solution is None:
  74          return None

  13  def _distances(data, gamma):

  23  def _shape(logd, weight=1.0, lower=0.0, upper=None, evaluations=None):
  24      z = logd - logd.mean()
  25
  26      def score(beta, source='solver'):
  27          w = np.exp(beta * (z - z.max()))
  28          w /= w.sum()
  29          value = float(weight / beta - w @ z)
  30          if evaluations is not None:
  31              evaluations.append(dict(beta=float(beta), score=value, source=source))
  32          return value
  33
  34      def recorded_result(beta, residual, boundary):
  35          # Supplement the same score function around its returned shape so the
  36          # criterion is readable. These calls occur after selection, never enter
  37          # root finding, and are identified separately from solver evaluations.
  38          if evaluations is not None:
  39              start = max(lower, beta * .5, 1e-8)
  40              stop = min(upper, beta * 1.5) if upper is not None else beta * 1.5
  41              for value in np.unique(np.r_[np.linspace(start, stop, 61), beta]):
  42                  score(float(value), source='diagnostic')
  43          return beta, residual, boundary
  44
  45      if np.ptp(logd) == 0:
  46          return None  # No finite shape root for an equal-valued sample.
  47      if lower > 0 and score(lower) <= 0:
  48          return recorded_result(float(lower), score(lower), True)
  49      if upper is None:
  50          upper = 2.0
  51          while score(upper) > 0:
  52              upper *= 2
  53              if not np.isfinite(upper):
  54                  return None
  55      elif score(upper) >= 0:
  56          return None  # WMLE needs an interior T1 zero, never a clamped boundary.
  57      beta = float(brentq(score, max(lower, 1e-8), upper, xtol=1e-11, rtol=2e-14))
  58      return recorded_result(beta, score(beta), False)

  79      def _calculate_r2(self, beta, eta, gamma_val):
  80          """Helper to calculate R-squared for any method"""
  81          if gamma_val >= self.data[0]: return 0.0
  82
  83          # Experimental
  84          x = np.log(self.data - gamma_val)
  85          y_exp = np.log(-np.log(1 - self._median_ranks()))
  86
  87          # Theoretical / Predicted
  88          # y = beta * x - beta * ln(eta)
  89          y_pred = beta * x - beta * np.log(eta)
  90
  91          # R^2 calculation
  92          ss_res = np.sum((y_exp - y_pred)**2)
  93          ss_tot = np.sum((y_exp - np.mean(y_exp))**2)
  94
  95          if ss_tot == 0: return 0.0
  96          return 1 - (ss_res / ss_tot)
