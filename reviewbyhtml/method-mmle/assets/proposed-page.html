原样本最小值确定γ，删除其中一个最小观测；对保留距离迭代求β，再恢复η。

  55          with np.errstate(divide='ignore', invalid='ignore', over='ignore'):
  56              logs = np.log(y)
  57              q = len(y)
  58              for iteration in range(1, max_iterations + 1):
  59                  weights = np.exp(beta * (logs - logs.max()))
  60                  weights /= weights.sum()
  61                  denominator = float(q * (weights @ logs))
  62                  next_beta = float((q + beta * logs.sum()) / denominator)
  63                  if not np.isfinite(next_beta) or next_beta <= 0:
  64                      break
  65                  difference = abs(next_beta - beta)
  66                  if iteration <= 3:
  67                      history.append(dict(iteration=iteration, shape_before=beta,
  68                                          shape_after=next_beta))

  70                      self.log_step(dict(phase='shape_iteration', iteration=iteration,
  71                                         shape_before=beta, shape_after=next_beta,
  72                                         shape_step=float(difference)))
  73                  beta = next_beta
  74                  if difference <= shape_absolute_step_tolerance:
