 135  def get_weight_j1(n: int) -> float:
 136      """
 137      Get J1 weight (median of W1) for sample size n.
 138
 139      J1 is used to compute the scale parameter beta.
 140      Paper definition: W1 = (1/n) * sum(log(1/(1-F(x_i)))), the summands are
 141      iid Exp(1), so W1 ~ Gamma(n, 1/n) and its exact median is
 142      gammaincinv(n, 0.5) / n (Cousineau 2009, Table 2 lists the same values).
 143      For n > 100 (outside the lookup table) we use this exact median.
 144      """
 145      if n in WEIGHT_TABLE_J1:
 146          return WEIGHT_TABLE_J1[n]
 147      if n < 1:
 148          return 0.5
 149      if n > 100:
 150          # Exact median of W1 ~ Gamma(n, 1/n)
 151          return float(gammaincinv(n, 0.5)) / n
 152      # Linear interpolation
 153      known_ns = sorted(WEIGHT_TABLE_J1.keys())
 154      for i in range(len(known_ns) - 1):
 155          n_low, n_high = known_ns[i], known_ns[i + 1]
 156          if n_low < n < n_high:
 157              w_low = WEIGHT_TABLE_J1[n_low]
 158              w_high = WEIGHT_TABLE_J1[n_high]
 159              t = (n - n_low) / (n_high - n_low)
 160              return w_low + t * (w_high - w_low)
 161      return WEIGHT_TABLE_J1[known_ns[-1]]

  23  def _shape(logd, weight=1.0, lower=0.0, upper=None, evaluations=None):
  24      z = logd - logd.mean()
  25
  26      def score(beta, source='solver'):
  27          w = np.exp(beta * (z - z.max()))
  28          w /= w.sum()
  29          value = float(weight / beta - w @ z)
  30          if evaluations is not None:
  31              evaluations.append(dict(beta=float(beta), score=value, source=source))
  32          return value
  33
  34      def recorded_result(beta, residual, boundary):
  35          # Supplement the same score function around its returned shape so the
  36          # criterion is readable. These calls occur after selection, never enter
  37          # root finding, and are identified separately from solver evaluations.
  38          if evaluations is not None:
  39              start = max(lower, beta * .5, 1e-8)
  40              stop = min(upper, beta * 1.5) if upper is not None else beta * 1.5
  41              for value in np.unique(np.r_[np.linspace(start, stop, 61), beta]):
  42                  score(float(value), source='diagnostic')
  43          return beta, residual, boundary
  44
  45      if np.ptp(logd) == 0:
  46          return None  # No finite shape root for an equal-valued sample.
  47      if lower > 0 and score(lower) <= 0:
  48          return recorded_result(float(lower), score(lower), True)
  49      if upper is None:
  50          upper = 2.0
  51          while score(upper) > 0:
  52              upper *= 2
  53              if not np.isfinite(upper):
  54                  return None
  55      elif score(upper) >= 0:
  56          return None  # WMLE needs an interior T1 zero, never a clamped boundary.
  57      beta = float(brentq(score, max(lower, 1e-8), upper, xtol=1e-11, rtol=2e-14))
  58      return recorded_result(beta, score(beta), False)

  88  def wmle_residuals(data, beta, gamma, weights):
  89      """Evaluate T1,T2 with the caller's J2/J3; beta/gamma may broadcast as grids."""
  90      x = np.asarray(data, dtype=float)
  91      b, g = np.broadcast_arrays(np.asarray(beta, float), np.asarray(gamma, float))
  92      d = x - g[..., None]
  93      if (x.ndim != 1 or len(x) < 2 or not np.all(np.isfinite(x))
  94              or np.any(~np.isfinite(b)) or np.any(b <= 0)
  95              or np.any(~np.isfinite(g)) or np.any(g < 0) or np.any(d <= 0)):
  96          raise ValueError("Invalid WMLE residual sample or parameter")
  97      t = np.log(d)
  98      z = t - t.mean(axis=-1, keepdims=True)
  99      powers = b[..., None] * z
 100      w = np.exp(powers - logsumexp(powers, axis=-1, keepdims=True))
 101      t1 = weights.get_weight_j2(len(x)) / b - np.sum(w * z, axis=-1)
 102      log_ratio = (t.mean(axis=-1) + logsumexp(powers, axis=-1)
 103                   - logsumexp((b[..., None] - 1) * z, axis=-1))
 104      j3 = np.asarray([weights.get_weight_j3(len(x), float(v)) for v in b.flat]).reshape(b.shape)
 105      t2 = np.mean(1 / d, axis=-1) * np.exp(log_ratio) - j3
 106      if b.ndim == 0:
 107          return float(t1), float(t2)
 108      return t1, t2

 198  def get_weight_j3(n: int, gamma: float) -> float:
 199      """
 200      Get J3 weight (median of W3) for sample size n and shape parameter gamma.
 201
 202      J3 is used in the location equation.
 203      Unlike J1 and J2, J3 depends on both n and gamma.
 204
 205      For gamma > 1, the asymptotic value is gamma / (gamma - 1).
 206
 207      Implementation follows R code approach:
 208      - Linear interpolation in gamma direction (0.1 resolution)
 209      - For n not in table, interpolate between nearest n values
 210      """
 211      global _J3_TABLE, _J3_GAMMA_VALUES
 212
 213      # Clamp gamma to [0.1, 5.0] (same as R code)
 214      gamma_clamped = max(0.1, min(5.0, gamma))
 215
 216      # Calculate asymptotic value for very large n
 217      if gamma_clamped > 1:
 218          asymp_val = gamma_clamped / (gamma_clamped - 1)
 219      else:
 220          asymp_val = float('inf')
 221
 222      # Helper function for gamma interpolation
 223      def interpolate_gamma(n_val):
 224          """Linear interpolation in gamma direction"""
 225          if n_val not in _J3_TABLE:
 226              return asymp_val if gamma_clamped > 1 else 2.0
 227
 228          gammas = sorted(_J3_TABLE[n_val].keys())
 229
 230          if gamma_clamped <= gammas[0]:
 231              return _J3_TABLE[n_val][gammas[0]]
 232          if gamma_clamped >= gammas[-1]:
 233              return _J3_TABLE[n_val][gammas[-1]]
 234
 235          # Linear interpolation
 236          for i in range(len(gammas) - 1):
 237              g_low, g_high = gammas[i], gammas[i + 1]
 238              if g_low <= gamma_clamped <= g_high:
 239                  j3_low = _J3_TABLE[n_val][g_low]
 240                  j3_high = _J3_TABLE[n_val][g_high]
 241                  t = (gamma_clamped - g_low) / (g_high - g_low)
 242                  return j3_low + t * (j3_high - j3_low)
 243
 244          return _J3_TABLE[n_val][gammas[-1]]
 245
 246      # Get available n values
 247      available_ns = sorted(_J3_TABLE.keys())
 248
 249      # n is in table
 250      if n in _J3_TABLE:
 251          return interpolate_gamma(n)
 252
 253      # n < minimum in table
 254      if n < available_ns[0]:
 255          return interpolate_gamma(available_ns[0])
 256
 257      # n > maximum in table (100)
 258      if n > available_ns[-1]:
 259          # Use asymptotic value for very large n
 260          if gamma_clamped > 1:
 261              return asymp_val
 262          else:
 263              return 2.0
 264
 265      # Interpolate between two nearest n values (bilinear interpolation)
 266      n_low, n_high = None, None
 267      for i in range(len(available_ns) - 1):
 268          if available_ns[i] < n < available_ns[i + 1]:
 269              n_low, n_high = available_ns[i], available_ns[i + 1]
 270              break
 271
 272      if n_low is None:
 273          # Fallback to closest value
 274          return interpolate_gamma(available_ns[-1])
 275
 276      # Get J3 values at both n values
 277      j3_low = interpolate_gamma(n_low)
 278      j3_high = interpolate_gamma(n_high)
 279
 280      # Linear interpolation in n direction
 281      t = (n - n_low) / (n_high - n_low)
 282      return j3_low + t * (j3_high - j3_low)

 134      """Numerically minimize T1**2+T2**2 over shape at a fixed location.
 135
 136      Search a shape grid including the J3 table knots, refine every bracketed
 137      local minimum, and compare all candidates. A T1 zero is an extra feasible
 138      candidate, not a constraint. This diagnostic does not run or replace the
 139      archived joint estimator. An endpoint result is a numerical boundary limit.
 140      """

 142      upper = float(weights.SHAPE_UPPER)
 143      if beta_grid is None:
 144          beta_grid = np.unique(np.r_[
 145              np.geomspace(1e-4, min(.1, upper * .5), 35),
 146              np.linspace(.1, min(5., upper * (1 - 1e-10)), 197),
 147              np.arange(.1, min(5., upper) + 1e-9, .1),
 148              np.linspace(min(5., upper * .5), upper * (1 - 1e-10), 101),
 149          ])

 151      grid = np.unique(grid[(grid > 0) & (grid < upper)])

 153          raise ValueError("Expected at least three finite shape grid points inside the search range")
 154

 156      values = t1 * t1 + t2 * t2
 157      candidates = [(float(grid[int(np.argmin(values))]), float(np.min(values)))]
 158

 160          a, b = wmle_residuals(data, float(beta), gamma, weights)
 161          return float(a * a + b * b)
 162
 163      minima = np.flatnonzero((values[1:-1] <= values[:-2]) & (values[1:-1] <= values[2:])) + 1

 165          result = minimize_scalar(objective, bounds=(grid[i - 1], grid[i + 1]),
